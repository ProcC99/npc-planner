```python
# src/npc_planner/ingest/cparse.py
from typing import Union

CValue = Union[int, str, list["CValue"], dict[str, "CValue"], "CMacroCall"]

class CParseError(ValueError):
    """Raised on unbalanced braces or malformed initialisers. Carries line and column."""

class CMacroCall:
    """An unexpanded function-like macro, e.g. LEVEL_UP_MOVE(7, MOVE_PECK)."""
    name: str
    args: list[CValue]

def parse_array_initializer(text: str, symbol: str) -> dict[str, dict[str, CValue]]:
    """Extract `TYPE symbol[] = { [KEY] = { .field = value, ... }, ... };` from text.

    Returns a mapping of KEY (as written, e.g. "SPECIES_SKARMORY") to a field dict.
    Raises CParseError if `symbol` is absent or braces are unbalanced.
    """

def parse_enum(text: str, enum_name: str | None = None) -> dict[str, int]:
    """Extract enum members and their resolved integer values, honouring implicit
    increment and explicit `= N` assignments."""

def parse_defines(text: str) -> dict[str, str]:
    """Extract object-like `#define NAME value` pairs from an unpreprocessed header."""
```

## Must-pass tests

1. Flat case: two entries, scalar int and identifier fields, parsed to the right keys.
2. Nested keyed initialiser: `.evolutions = { .method = EVO_LEVEL, .param = 30 }` -> nested dict.
3. Positional array: three-element `.abilities` list preserved in order.
4. Macro preservation: `LEVEL_UP_MOVE(7, MOVE_PECK)` -> `CMacroCall(name="LEVEL_UP_MOVE", args=[7, "MOVE_PECK"])`.
5. String literal with escaped quote and adjacent-literal concatenation.
6. `#line 0` directives and `/* block */` plus `// line` comments are stripped without corrupting surrounding values.
7. Unbalanced brace raises `CParseError` with a line number.
8. Missing symbol raises `CParseError` naming the symbol.
9. `parse_enum` resolves implicit increment: `A, B = 5, C` -> `{A: 0, B: 5, C: 6}`.
10. Property test: for any generated well-formed initialiser, `parse_array_initializer` returns exactly the generated entry count.

## Out of scope — do NOT implement

- Any file reading — this module takes strings only
- Any mapping of parsed dicts onto domain schemas — that is the per-domain extractors
- Evaluating arithmetic or resolving `#define` substitution
